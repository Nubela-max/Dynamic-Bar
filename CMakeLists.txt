cmake_minimum_required(VERSION 3.21)
project(dynamic-bar VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 DBus)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_executable(dynamic-bar src/main.cpp src/shellcontroller.cpp src/shellcontroller.h src/configmanager.cpp src/configmanager.h)
qt_add_qml_module(dynamic-bar URI DynamicBar VERSION 1.0 QML_FILES qml/Main.qml qml/components/PillButton.qml qml/components/ModuleChip.qml qml/panels/Launcher.qml qml/panels/ControlCenter.qml qml/panels/Clipboard.qml qml/panels/Settings.qml)
target_link_libraries(dynamic-bar PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::DBus)
install(TARGETS dynamic-bar RUNTIME DESTINATION bin)
install(FILES packaging/dynamic-bar.desktop DESTINATION share/applications)
install(FILES packaging/dynamic-bar-autostart.desktop DESTINATION share/wayland-sessions)
